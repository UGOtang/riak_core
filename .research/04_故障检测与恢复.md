# 故障检测与恢复 - Riak Core 的自我修复能力

> 📖 **对应DDIA章节**: 第八章《分布式系统的麻烦》中的"故障检测"与"恢复"

## 🎯 核心问题：当节点崩溃或网络中断时，如何保证数据安全？

想象你在管理一个分布式银行：
- 某个分行的服务器突然断电
- 客户的钱还能取出来吗？
- 服务器恢复后，如何同步丢失的交易？

### 传统方案的问题

| 方案 | 问题 |
|------|------|
| 主从切换 | 需要外部协调者，增加复杂度 |
| 多数派投票 | 网络分区时无法工作 |
| 静态故障转移 | 需要人工干预 |

### Riak Core 的答案：**Hinted Handoff + Active Anti-Entropy**

---

## 🧪 实验一：Hinted Handoff - 临时托管

### 核心思想：别人生病，你替他保管工作

```
正常情况:
  Node1 (分区A) ← 客户端写入数据
  Node2 (分区B) ← 副本
  Node3 (分区C) ← 副本

Node1 宕机时:
  Node1 (X)
  Node2 (分区B) ← 客户端写入数据
       └── 同时保存分区A的"提示" (Hint)
  Node3 (分区C) ← 副本

Node1 恢复后:
  Node2 把"提示"数据发送回 Node1
  Node1 恢复正常工作
```

### Hinted Handoff 实现

```erlang
%% 文件: src/riak_core_handoff_manager.erl

-record(handoff_status, {
    mod_src_tgt :: {module(), partition(), partition()},
    vnode_pid :: pid(),
    transport_pid :: pid(),
    type :: repair | resize | hinted,
    ...
}).

%% 启动 hinted handoff
start_hinted_handoff(Partition, TargetNode, Data) ->
    %% 1. 创建 handoff 进程
    {ok, Sender} = riak_core_handoff_sender:start_link(
        hinted, Partition, TargetNode, Data
    ),
    %% 2. 记录 handoff 状态
    add_outbound(hinted, Partition, TargetNode, Sender).
```

## 🧪 实验二：Active Anti-Entropy (AAE) - 主动修复

### 问题：Hinted Handoff 不够用

```
场景1: Hint 丢失
  Node2 保存了 hint，但自己也崩溃了

场景2: 长时间宕机
  Node1 宕机时间太长，hint 数据过多

场景3: 数据静默损坏
  磁盘错误导致数据损坏，但没有报告
```

### AAE 解决方案：定期比对数据

```erlang
%% 文件: src/hashtree.erl

%% 核心思想：使用 Merkle Tree 快速比对数据差异

%% Hash Tree 结构
%% level 1:  1个根节点，包含整个数据集的 hash
%% level 2:  4个节点，每个包含 1/4 数据的 hash
%% level 3: 16个叶子节点，每个包含具体数据段的 hash

%% 比对过程：
%% 1. 比对根节点 hash
%%    - 如果相同，数据一致，结束
%%    - 如果不同，继续向下
%% 2. 比对 level 2 的 4 个节点
%%    - 找出不同的节点，继续向下
%% 3. 最终定位到具体的数据段
%% 4. 只同步不同的数据
```

### Hash Tree 实现

```erlang
%% 文件: src/hashtree.erl

%% Hash Tree 的核心操作

%% 插入数据
insert(Key, Hash, Tree) ->
    %% 1. 计算数据属于哪个 segment
    Segment = key_to_segment(Key),
    %% 2. 写入 segment 存储
    segment_insert(Segment, Key, Hash),
    %% 3. 标记 segment 为 dirty
    mark_dirty(Segment),
    ok.

%% 更新 Hash Tree
update(Tree) ->
    %% 1. 遍历所有 dirty segments
    DirtySegments = get_dirty_segments(),
    %% 2. 计算每个 dirty segment 的 hash
    SegmentHashes = [{S, hash_segment(S)} || S <- DirtySegments],
    %% 3. 更新叶子节点
    update_leaves(SegmentHashes),
    %% 4. 自底向上更新树
    update_tree_bottom_up(),
    ok.

%% 比对两棵树
compare(LocalTree, RemoteTree, GetRemoteBucket, DiffCallback) ->
    %% 从根节点开始递归比对
    compare_level(1, 0, LocalTree, RemoteTree, GetRemoteBucket, DiffCallback).

compare_level(Level, Bucket, Local, Remote, GetRemote, Callback) ->
    LocalHash = get_bucket_hash(Local, Level, Bucket),
    RemoteHash = GetRemote(Level, Bucket),
    
    case LocalHash =:= RemoteHash of
        true -> 
            %% 相同，无需处理
            ok;
        false ->
            case is_leaf(Level) of
                true ->
                    %% 叶子节点，获取具体差异
                    Diff = get_key_diffs(Bucket, Local, Remote),
                    Callback(Diff);
                false ->
                    %% 内部节点，递归检查子节点
                    Children = get_children(Bucket),
                    [compare_level(Level+1, C, Local, Remote, GetRemote, Callback)
                     || C <- Children]
            end
    end.
```

```
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  Node1                           Node2                      │
│    │                               │                        │
│    │──── 1. 请求根节点 hash ───────>│                        │
│    │<─── 返回 hash(root) ─────────│                        │

│    │                               │                        │
│    │  发现 h1, h3 不同             │                        │
│    │                               │                        │
│    │──── 3. 请求 segment 差异 ────>│                        │
│    │<─── 返回 segment keys ───────│                        │
│    │                               │                        │
│    │  比对 keys，找出差异           │                        │
│    │                               │                        │
│    │──── 4. 请求缺失数据 ─────────>│                        │
│    │<─── 返回数据 ────────────────│                        │
│    │                               │                        │

```

## 🧪 实验三：Read Repair - 读时修复

### 核心思想：读数据时顺便修复

```erlang
%% 读取数据的流程
read(Key, N, R) ->
    %% 1. 获取优先列表
    PrefList = get_apl(Key, N),
    
    %% 2. 向 R 个节点发起读取
    Responses = [VNode:get(Key) || VNode <- lists:sublist(PrefList, R)],
    
    %% 3. 等待 R 个响应
    {ok, Values} = wait_for_responses(Responses, R),
    
    %% 4. 检测冲突
    case has_conflicts(Values) of
        false -> 
            %% 数据一致，返回
            {ok, hd(Values)};
        true ->
            %% 有冲突，触发 read repair
            Resolved = resolve_conflicts(Values),
            %% 异步修复所有副本
            spawn(fun() -> repair_all_replicas(Key, Resolved, PrefList) end),
            {ok, Resolved}
    end.

%% 修复所有副本
repair_all_replicas(Key, Value, PrefList) ->
    [{VNode, put(Key, Value)} || VNode <- PrefList].
```

## 🧪 实验四：Handoff - 数据迁移

### Handoff 类型

```erlang
-type ho_type() :: hinted | ownership | repair | resize.

%% hinted:    Hinted Handoff，节点恢复后归还数据
%% ownership: 所有权转移，节点离开或加入
%% repair:    修复性 Handoff，手动触发
%% resize:    Ring 扩缩容
```

### Handoff 实现架构

```erlang
%% 文件: src/riak_core_handoff_manager.erl

%% Handoff Manager 协调所有 Handoff 操作
-record(state, {
    excl,          %% 排除列表（禁止 handoff 的分区）
    handoffs=[]    %% 正在进行的 handoff 列表
}).

%% 添加出站 Handoff
add_outbound(Type, Module, SrcIdx, TargetIdx, Node, VnodePid, Opts) ->
    case application:get_env(riak_core, disable_outbound_handoff) of
        {ok, true} ->
            {error, max_concurrency};
        _ ->
            %% 检查并发限制
            case check_concurrency() of
                ok ->
                    %% 启动 sender 进程
                    start_handoff_sender(Type, Module, SrcIdx, TargetIdx, Node, VnodePid);
                {error, Reason} ->
                    {error, Reason}
            end
    end.
```

### Handoff Sender 实现

```erlang
%% 文件: src/riak_core_handoff_sender.erl

%% Handoff Sender 负责发送数据
init([Type, Partition, TargetNode, Module, Opts]) ->
    %% 1. 连接目标节点
    {ok, Socket} = connect(TargetNode),
    
    %% 2. 发送 Handoff 请求
    send_handoff_req(Socket, Partition, Module),
    
    %% 3. 获取 vnode 数据迭代器
    Iterator = Module:fold(Partition, fun fold_fun/3, []),
    
    %% 4. 流式发送数据
    send_data(Socket, Iterator),
    
    %% 5. 发送结束消息
    send_eob(Socket),
    
    {stop, normal, State}.

fold_fun(K, V, Acc) ->
    %% 编码数据
    Bin = encode_kv(K, V),
    [{K, V, Bin} | Acc].

send_data(Socket, [{K, V, Bin} | Rest]) ->
    %% 发送数据包
    ok = gen_tcp:send(Socket, Bin),
    %% 更新进度
    update_progress(K),
    %% 继续发送
    send_data(Socket, Rest);
send_data(_Socket, []) ->
    ok.
```

### Handoff Receiver 实现

```erlang
%% 文件: src/riak_core_handoff_receiver.erl

%% Handoff Receiver 负责接收数据
init([Partition, Module]) ->
    %% 1. 获取或创建目标 vnode
    {ok, VNode} = get_or_create_vnode(Partition, Module),
    
    %% 2. 进入接收循环
    receive_loop(VNode, Module).

receive_loop(VNode, Module) ->
    receive
        {tcp, Socket, Data} ->
            %% 解码数据
            {K, V} = decode_kv(Data),
            %% 写入 vnode
            Module:handle_handoff_data(K, V, VNode),
            %% 继续接收
            receive_loop(VNode, Module);
            
        {tcp_closed, Socket} ->
            %% Handoff 完成
            Module:handoff_finished(VNode),
            {stop, normal};
            
        {error, Reason} ->
            {stop, Reason}
    end.
```

## 🧪 实验五：故障场景处理

### 场景1：单节点故障

```
初始状态:
  Partition A: [Node1(primary), Node2, Node3]
  数据: {key1, value1}

故障:
  Node1 宕机

处理:
  1. Node Watcher 检测到 Node1 down
  2. 更新 Ring: Node1 → down
  3. 后续写入:
     - 客户端写入 Node2 (fallback)
     - Node2 保存 hint for Node1
  4. Node1 恢复后:
     - Node Watcher 检测到 Node1 up
     - Node2 触发 hinted handoff
     - 数据归还给 Node1
```

### 场景2：网络分区

```
初始状态:
  Cluster: [Node1, Node2, Node3, Node4, Node5]

分区发生:
  分区1: [Node1, Node2]
  分区2: [Node3, Node4, Node5]

处理:
  1. Node Watcher 在两边互相检测到对方 "down"
  2. 分区1 继续提供服务:
     - 写入数据到可用节点
     - 保存 hints
  3. 分区2 同样继续服务
  4. 分区恢复后:
     - 两边合并数据
     - 使用 vector clock 解决冲突
     - 可能产生 siblings
```

### 场景3：数据损坏

```
发现:
  AAE 检查发现 Node1 和 Node2 数据不一致

处理:
  1. AAE 比对 Hash Tree
  2. 定位损坏的数据段
  3. 从其他副本读取正确数据
  4. 修复损坏的副本
  5. 更新 Hash Tree
```

## 📊 故障恢复策略对比

| 策略 | 触发时机 | 适用场景 | 开销 |
|------|----------|----------|------|
| Hinted Handoff | 节点恢复时 | 短期故障 | 低 |
| Read Repair | 读取时 | 发现不一致 | 中 |
| AAE | 后台定期 | 长期故障/数据损坏 | 低（分散） |
| 手动修复 | 人工触发 | 紧急情况 | 高 |
