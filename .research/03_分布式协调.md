# 分布式协调 - Riak Core 的集群大脑

> 📖 **对应DDIA章节**: 第九章《一致性与共识》与第八章《分布式系统的麻烦》

## 🎯 核心问题：如何让所有节点对"谁是集群成员"达成一致？

想象你在组织一个分布式会议：
- 参会者随时可能加入或离开
- 网络可能中断
- 每个人都要知道"谁在会议室里"

### 传统方案的问题

| 方案 | 问题 |
|------|------|
| 中心化协调者 | 单点故障 |
| 全对全通信 | O(n²) 消息复杂度 |
| Paxos/Raft | 实现复杂，性能开销大 |

### Riak Core 的答案：**Gossip 协议 + Claimant 机制**

---

## 🧪 实验一：Gossip 协议 - 最终一致性的传播

### 核心思想：像八卦一样传播信息

```
时间 T0:
  Node1 知道: {Node1, Node2}
  Node2 知道: {Node1, Node2}
  Node3 知道: {Node3}

时间 T1 (Gossip 后):
  Node1 随机选择 Node2 交换信息
  Node2 随机选择 Node3 交换信息
  
时间 T2 (Gossip 后):
  所有节点都知道: {Node1, Node2, Node3}
```

### Gossip 实现

```erlang
%% 文件: src/riak_core_gossip.erl

%% 随机 Gossip: 随机选择一个节点交换状态
random_gossip(Ring) ->
    case riak_core_ring:random_other_active_node(Ring) of
        no_node -> 
            ok;  %% 单节点集群
        RandomNode ->
            send_ring(node(), RandomNode)
    end.

%% 发送本地环状态到目标节点
send_ring(FromNode, ToNode) ->
    gen_server:cast({?MODULE, FromNode}, {send_ring_to, ToNode}).
```

### 递归 Gossip: 更高效的传播

```erlang
%% 使用二叉树结构传播，log(N) 轮即可覆盖全集群
recursive_gossip(Ring, Node) ->
    Nodes = riak_core_ring:active_members(Ring),
    Tree = riak_core_util:build_tree(2, Nodes, [cycles]),  %% 二叉树
    Children = orddict:fetch(Node, Tree),
    _ = [send_ring(node(), OtherNode) || OtherNode <- Children],
    ok.
```

### Gossip 流量控制

```erlang
%% 默认限流: 每10秒最多45次 Gossip
-define(DEFAULT_LIMIT, {45, 10000}).

handle_cast({send_ring_to, _Node}, State=#state{gossip_tokens=0}) ->
    %% 没有令牌了，忽略本次 Gossip
    {noreply, State};
handle_cast({send_ring_to, Node}, State) ->
    %% 发送环状态，消耗一个令牌
    {ok, MyRing} = riak_core_ring_manager:get_raw_ring(),
    gen_server:cast({?MODULE, Node}, {reconcile_ring, MyRing}),
    {noreply, State#state{gossip_tokens=State#state.gossip_tokens-1}}.
```

## 🧪 实验二：Ring Reconciliation - 环的合并

### 问题：不同节点可能有不同的环视图

```
Node1 的视图: {Node1, Node2, Node3}
Node2 的视图: {Node1, Node2}  (还没收到 Node3 加入的消息)
Node3 的视图: {Node1, Node2, Node3}

需要合并策略！
```

### 基于 Vector Clock 的环版本

```erlang
%% 文件: src/riak_core_ring.erl

-record(chstate_v2, {
    nodename :: term(),          % 节点名
    vclock   :: vclock:vclock(), % 环的版本向量！
    chring   :: chash:chash(),   % 一致性哈希环
    meta     :: dict(),          % 元数据
    clustername :: {term(), term()},
    members  :: [{node(), {member_status(), vclock:vclock(), [...]}}],
    ...
}).
```

### 环合并算法

```erlang
%% 文件: src/riak_core_gossip.erl

reconcile(Ring0, [OtherRing0]) ->
    {Ring, OtherRing} = riak_core_ring:reconcile_names(Ring0, OtherRing0),
    
    %% 1. 检查是否来自同一集群
    WrongCluster = (riak_core_ring:cluster_name(Ring) /=
                    riak_core_ring:cluster_name(OtherRing)),
    
    %% 2. 检查发送者的状态
    PreStatus = riak_core_ring:member_status(Members, OtherNode),
    IgnoreGossip = (WrongCluster or
                    (PreStatus =:= invalid) or
                    (PreStatus =:= down)),
    
    %% 3. 合并环
    {Changed, Ring2} = 
        case IgnoreGossip of
            true -> {false, Ring};
            false -> riak_core_ring:reconcile(OtherRing, Ring)
        end,
    
    %% 4. 返回结果
    case Changed of
        new_ring -> {reconciled_ring, Ring3};
        _ -> ignore
    end.
```

### 图示：环合并过程

```
步骤1: 接收到其他节点的环
  ┌─────────────────────────────────────┐
  │ Node1 收到 Node2 的环               │
  │ LocalRing:  vclock=[n1:3, n2:2]     │
  │ RemoteRing: vclock=[n1:2, n2:3]     │
  └─────────────────────────────────────┘
           │
           ↓
步骤2: 比较版本向量
  ┌─────────────────────────────────────┐
  │ descends(Local, Remote)? = false    │
  │ descends(Remote, Local)? = false    │
  │ 结论: 并发！需要合并                 │
  └─────────────────────────────────────┘
           │
           ↓
步骤3: 合并两个环
  ┌─────────────────────────────────────┐
  │ MergedVC = merge(Local, Remote)     │
  │        = [n1:3, n2:3]               │
  │ 取每个节点的最大计数                 │
  └─────────────────────────────────────┘
           │
           ↓
步骤4: 更新本地环
  ┌─────────────────────────────────────┐
  │ LocalRing.vclock = [n1:3, n2:3]     │
  │ 触发 ring_changed 事件               │
  └─────────────────────────────────────┘
```

## 🧪 实验三：Claimant - 集群协调者

### 问题：谁来做决策？

```
场景：Node3 要加入集群
- Node1 和 Node2 都收到了请求
- 谁来决定分区如何重新分配？
- 如果两人同时决定，会冲突吗？
```

### Claimant 机制：唯一的决策者

```erlang
%% 文件: src/riak_core_claimant.erl

%% Claimant 是环中记录的一个特殊节点
%% 它是第一个加入集群的节点，或者被指定的继任者

%% 检查当前节点是否是 Claimant
is_claimant(Ring) ->
    riak_core_ring:claimant(Ring) =:= node().

%% Claimant 的职责
%% 1. 处理成员变更请求
%% 2. 计算新的分区分配
%% 3. 推进集群状态机
```

### 成员状态机

```erlang
-type member_status() :: joining | valid | invalid | leaving | exiting | down.

状态转换图：

        joining ──(claimant批准)──> valid
           │                          │
           │                    (发起离开)
           │                          │
           │                          ↓
           │                      leaving
           │                          │
           │                    (handoff完成)
           │                          │
           │                          ↓
           └────────────────────> exiting
                                      │
                                (节点关闭)
                                      │
                                      ↓
                                   invalid
                                      
        valid ──(检测到宕机)──> down
          │                         │
          │                   (节点恢复)
          │                         │
          └─────────────────────────┘
```

### Claimant 的核心逻辑

```erlang
%% 文件: src/riak_core_claimant.erl

ring_changed(Ring, NewRing) ->
    case is_claimant(NewRing) of
        true ->
            %% 我是 Claimant，执行协调逻辑
            maybe_update_ring(Ring, NewRing);
        false ->
            %% 不是 Claimant，忽略
            ok
    end.

maybe_update_ring(OldRing, NewRing) ->
    %% 1. 检查是否有待处理的变更
    Changes = riak_core_ring:pending_changes(NewRing),
    
    %% 2. 执行变更
    case Changes of
        [] -> ok;
        _ -> 
            %% 计算新的分区分配
            Plan = plan_changes(NewRing, Changes),
            %% 应用变更
            apply_plan(Plan)
    end.
```

## 🧪 实验四：成员管理 - 加入与离开

### 加入集群

```erlang
%% 步骤1: 新节点向任意节点发送加入请求
riak_core:join(Node, ClusterNode) ->
    %% ClusterNode 会转发给 Claimant
    gen_server:cast({?MODULE, ClusterNode}, {join, Node}).

%% 步骤2: Claimant 处理加入请求
handle_join(Node, Ring) ->
    %% 更新环：标记 Node 为 joining
    Ring2 = riak_core_ring:add_member(Node, Ring, Node),
    %% 分配一些分区给新节点
    Ring3 = claim_partitions(Ring2, Node),
    %% 广播新环
    riak_core_gossip:distribute_ring(Ring3).

%% 步骤3: 新节点收到环，启动 Handoff 接收数据
handle_info({ring_update, NewRing}, State) ->
    %% 检查是否有分区需要接收
    MyNewPartitions = get_new_partitions(NewRing),
    %% 启动 Handoff
    [start_handoff(P) || P <- MyNewPartitions].
```

### 离开集群

```erlang
%% 步骤1: 节点发起离开请求
riak_core:leave() ->
    riak_core_claimant:leave_member(node()).

%% 步骤2: Claimant 处理离开请求
handle_leave(Node, Ring) ->
    %% 标记 Node 为 leaving
    Ring2 = riak_core_ring:leave_member(node(), Ring, Node),
    %% 计算分区迁移计划
    Plan = plan_transfers(Ring2, Node),
    %% 广播新环
    riak_core_gossip:distribute_ring(Ring2).

%% 步骤3: 离开节点 Handoff 数据到其他节点
handle_transfer(Partition, TargetNode) ->
    %% 发送所有数据
    send_all_data(Partition, TargetNode),
    %% 完成后通知 Claimant
    riak_core_ring:handoff_complete(node(), Partition, TargetNode).

%% 步骤4: 所有 Handoff 完成后，标记为 exiting
all_handoffs_complete(Ring, Node) ->
    Ring2 = riak_core_ring:exit_member(node(), Ring, Node),
    riak_core_gossip:distribute_ring(Ring2).
```

## 🧪 实验五：故障检测

### Node Watcher: 心跳监控

```erlang
%% 文件: src/riak_core_node_watcher.erl

-record(state, {
    status = up,              %% 本节点状态
    services = [],            %% 提供的服务
    health_checks = [],       %% 健康检查
    peers = [],               %% 集群其他节点
    avsn = 0,                 %% 版本号
    bcast_tref                %% 广播定时器
}).

%% 定期广播本节点状态
schedule_broadcast(State) ->
    Interval = app_helper:get_env(riak_core, gossip_interval),
    Tref = erlang:send_after(Interval, self(), broadcast),
    State#state { bcast_tref = Tref }.

broadcast(Nodes, State) ->
    case State#state.status of
        up -> Msg = {up, node(), State#state.services};
        down -> Msg = {down, node()}
    end,
    gen_server:abcast(Nodes, ?MODULE, Msg).
```

### 健康检查机制

```erlang
%% 文件: src/riak_core_node_watcher.erl

-record(health_check, {
    state = 'waiting' :: 'waiting' | 'checking' | 'suspend',
    callback :: {atom(), atom(), [any()]},  %% 检查函数
    service_pid :: pid(),
    checking_pid :: pid() | undefined,
    health_failures = 0 :: non_neg_integer(),
    callback_failures = 0 :: non_neg_integer(),
    interval_tref,
    check_interval = 60000 :: timeout(),
    max_callback_failures = 3,
    max_health_failures = 1
}).

%% 健康检查状态机
health_fsm(waiting, check_health, Service, InCheck) ->
    %% 启动健康检查进程
    start_health_check(Service, InCheck);

health_fsm(checking, {result, Pid, true}, Service, InCheck) ->
    %% 检查通过，重置失败计数
    {ok, waiting, InCheck#health_check{health_failures=0}};

health_fsm(checking, {result, Pid, false}, Service, InCheck) ->
    %% 检查失败，增加失败计数
    Fails = InCheck#health_check.health_failures + 1,
    case Fails >= InCheck#health_check.max_health_failures of
        true -> 
            %% 标记服务为 down
            {down, waiting, InCheck#health_check{health_failures=Fails}};
        false ->
            {ok, waiting, InCheck#health_check{health_failures=Fails}}
    end.
```

### 故障检测流程图

```
┌─────────────────────────────────────────────────────────────┐
│                     Node Watcher                            │
├─────────────────────────────────────────────────────────────┤
│  定时器 ──> broadcast() ──> 向所有 peers 发送 {up, Services} │
│                                                              │
│  收到 {up, Node, Services} ──> 更新 ETS 表                   │
│  收到 {down, Node} ──> 从 ETS 表移除                         │
│                                                              │
│  检测到 nodedown ──> 标记节点为 down                         │
└─────────────────────────────────────────────────────────────┘
         │
         ↓
┌─────────────────────────────────────────────────────────────┐
│                   Ring Manager                              │
├─────────────────────────────────────────────────────────────┤
│  收到节点 down ──> 更新环状态                                │
│                   触发 fallback 节点接管                     │
│                   可能触发 repair                            │
└─────────────────────────────────────────────────────────────┘
```

## 🎨 实际应用示例

### 场景：检测并处理节点故障

```erlang
%% 1. Node Watcher 检测到节点宕机
handle_info({nodedown, Node}, State) ->
    %% 从 ETS 表移除节点
    node_down(Node, State),
    {noreply, update_avsn(State)};

%% 2. Ring Manager 更新环状态
handle_info({ring_update, NewRing}, State) ->
    %% 检查是否有节点状态变化
    DownNodes = riak_core_ring:down_members(NewRing),
    %% 对每个宕机节点，触发 fallback
    [trigger_fallback(Node, NewRing) || Node <- DownNodes].

%% 3. 启动 fallback vnode
trigger_fallback(Node, Ring) ->
    %% 找到宕机节点的分区
    Partitions = riak_core_ring:indices(Ring, Node),
    %% 在其他节点启动 fallback vnode
    [start_fallback_vnode(P) || P <- Partitions].
```

### 场景：集群扩容

```erlang
%% 1. 新节点启动，加入集群
riak_core:join(NewNode, ExistingNode).

%% 2. Claimant 收到请求，分配分区
handle_cast({join, Node}, State) ->
    Ring = State#state.ring,
    %% 计算要迁移的分区
    PartitionsToTransfer = calculate_transfers(Ring, Node),
    %% 更新环状态
    Ring2 = riak_core_ring:transfer_node(PartitionsToTransfer, Node, Ring),
    %% 广播新环
    riak_core_gossip:distribute_ring(Ring2).

%% 3. 相关节点启动 Handoff
handle_info({ring_update, NewRing}, State) ->
    %% 检查是否有分区需要迁出
    Transfers = get_outbound_transfers(NewRing),
    [start_handoff(Partition, TargetNode) || {Partition, TargetNode} <- Transfers].
```

## 📊 Gossip 协议的优势与劣势

### 优势

```
1. 可扩展性
   - 消息复杂度: O(n) 而非 O(n²)
   - 每个节点只与少数节点通信

2. 容错性
   - 单个节点故障不影响整体
   - 信息最终会传播到所有节点

3. 简单性
   - 实现简单
   - 无需复杂的共识协议
```

### 劣势

```
1. 最终一致性
   - 信息传播有延迟
   - 短期内节点视图可能不一致

2. 带宽开销
   - 定期广播状态
   - 即使没有变更也要发送

3. 冲突处理
   - 并发变更可能产生冲突
   - 需要额外的冲突解决机制
```

## 🔬 深度分析：为什么 Riak 不用 Paxos/Raft？

### 设计权衡

```
Paxos/Raft:
┌─────────────────────────────────────┐
│ 优势: 强一致性                       │
│ 劣势: 需要多数派节点存活             │
│       网络分区时可能不可用           │
│       实现复杂                       │
└─────────────────────────────────────┘

Gossip + Claimant:
┌─────────────────────────────────────┐
│ 优势: 高可用性                       │
│       网络分区时仍可工作             │
│       实现简单                       │
│ 劣势: 最终一致性                     │
│       状态收敛需要时间               │
└─────────────────────────────────────┘
```

### Riak 的选择依据

```erlang
%% Riak 的设计哲学：
%% 1. 可用性优先 (AP 系统)
%% 2. 接受最终一致性
%% 3. 简单可维护

%% 这意味着：
%% - 牺牲强一致性，换取高可用
%% - 接受短期内集群视图不一致
%% - 用简单的 Gossip 替代复杂的共识
```
